function(nmc_add_test name)
  add_executable(${name} ${ARGN})
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_compile_options(${name} PRIVATE /W4 /WX /EHsc /permissive-)
  target_link_libraries(${name} PRIVATE near_memory_compute ws2_32)
  if(NMC_ASAN)
    target_compile_options(${name} PRIVATE /fsanitize=address /Zi)
    target_link_options(${name} PRIVATE /fsanitize=address)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endfunction()

nmc_add_test(nmc_test_engine test_engine.cpp)
nmc_add_test(nmc_test_codec test_codec.cpp)
nmc_add_test(nmc_test_persist test_persist.cpp)
nmc_add_test(nmc_test_property test_property.cpp)
nmc_add_test(nmc_test_concurrency test_concurrency.cpp)
nmc_add_test(nmc_test_adversarial test_adversarial.cpp)
nmc_add_test(nmc_test_system test_system.cpp)
nmc_add_test(nmc_test_distributed test_distributed.cpp)
